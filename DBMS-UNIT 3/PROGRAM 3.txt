DECLARE
    v_name   EMP.ENAME%TYPE;
    v_salary EMP.BASICSAL%TYPE;
BEGIN
    v_name := '&Enter_Employee_Name';

    SELECT BASICSAL
    INTO v_salary
    FROM EMP
    WHERE ENAME = v_name;

    DBMS_OUTPUT.PUT_LINE('Employee Name: ' || v_name);
    DBMS_OUTPUT.PUT_LINE('Basic Salary: ' || v_salary);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee does not exist in EMP table.');
END;
/
