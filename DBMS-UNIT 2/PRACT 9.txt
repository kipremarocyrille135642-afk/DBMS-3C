SET SERVEROUTPUT ON;

DECLARE
  -- Cursor selecting employee records, sorted by salary (descending)
  CURSOR emp_cur IS
    SELECT EMPNO, ENAME, DEPTNO, SAL
    FROM   EMP
    ORDER BY SAL DESC;

  v_emp_rec  emp_cur%ROWTYPE;
  v_count    NUMBER := 0;

BEGIN
  OPEN emp_cur;

  LOOP
    FETCH emp_cur INTO v_emp_rec;
    EXIT WHEN emp_cur%NOTFOUND;

    DBMS_OUTPUT.PUT_LINE('Emp No : ' || v_emp_rec.EMPNO);
    DBMS_OUTPUT.PUT_LINE('Name   : ' || v_emp_rec.ENAME);
    DBMS_OUTPUT.PUT_LINE('Dept   : ' || v_emp_rec.DEPTNO);
    DBMS_OUTPUT.PUT_LINE('Salary : ' || v_emp_rec.SAL);
    DBMS_OUTPUT.PUT_LINE('------------------------------');

    v_count := v_count + 1;

  END LOOP;

  IF v_count = 0 THEN
    DBMS_OUTPUT.PUT_LINE('No records found in EMP table.');
  ELSE
    DBMS_OUTPUT.PUT_LINE('Total ' || v_count || ' record(s) displayed in descending order of salary.');
  END IF;

  CLOSE emp_cur;

EXCEPTION
  WHEN OTHERS THEN
    IF emp_cur%ISOPEN THEN
      CLOSE emp_cur;
    END IF;
    DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END;
/
