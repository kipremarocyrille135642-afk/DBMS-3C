SET SERVEROUTPUT ON;

DECLARE
  v_empno   EMPLOYEE.EMPNO%TYPE := &emp_no;
  v_hike    NUMBER := 500;
BEGIN
  -- The UPDATE statement itself is an implicit cursor (referred to as SQL)
  UPDATE EMPLOYEE
  SET    SAL = SAL + v_hike
  WHERE  EMPNO = v_empno;

  IF SQL%FOUND THEN
    DBMS_OUTPUT.PUT_LINE('Salary updated successfully for employee number ' ||
                          v_empno || '. Rows affected: ' || SQL%ROWCOUNT);
    COMMIT;
  ELSE
    DBMS_OUTPUT.PUT_LINE('No employee found with employee number ' || v_empno || '.');
  END IF;

EXCEPTION
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END;
/
