SET SERVEROUTPUT ON;

-- ============================================================
-- CREATE THE PROCEDURE (no parameters)
-- ============================================================
CREATE OR REPLACE PROCEDURE update_emp_sal
IS
  v_count NUMBER;
BEGIN
  UPDATE EMP
  SET    SAL = SAL + (SAL * 10 / 100);

  v_count := SQL%ROWCOUNT;

  IF v_count > 0 THEN
    DBMS_OUTPUT.PUT_LINE(v_count || ' employee record(s) updated successfully with a 10% salary raise.');
    COMMIT;
  ELSE
    DBMS_OUTPUT.PUT_LINE('No records found in EMP table to update.');
  END IF;

EXCEPTION
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END update_emp_sal;
/


-- ============================================================
-- CALL THE PROCEDURE FROM A PL/SQL BLOCK
-- ============================================================
BEGIN
  update_emp_sal;
END;
/
