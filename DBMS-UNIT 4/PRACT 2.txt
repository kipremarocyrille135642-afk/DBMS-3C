SET SERVEROUTPUT ON;

-- ============================================================
-- CREATE THE PROCEDURE (with IN parameters)
-- ============================================================
CREATE OR REPLACE PROCEDURE raise_salary (
  p_deptno  IN EMP.DEPTNO%TYPE,
  p_percent IN NUMBER
)
IS
  v_count NUMBER := 0;
BEGIN
  UPDATE EMP
  SET    SAL = SAL + (SAL * p_percent / 100)
  WHERE  DEPTNO = p_deptno;

  v_count := SQL%ROWCOUNT;

  IF v_count > 0 THEN
    DBMS_OUTPUT.PUT_LINE(v_count || ' employee(s) in department ' || p_deptno ||
                          ' received a ' || p_percent || '% salary raise.');
    COMMIT;
  ELSE
    DBMS_OUTPUT.PUT_LINE('No employees found in department ' || p_deptno || '.');
  END IF;

EXCEPTION
  WHEN OTHERS THEN
    DBMS_OUTPUT.PUT_LINE('An error occurred: ' || SQLERRM);
END raise_salary;
/


-- ============================================================
-- CALL THE PROCEDURE FROM A PL/SQL BLOCK
-- ============================================================
DECLARE
  v_deptno  EMP.DEPTNO%TYPE := &dept_no;
  v_percent NUMBER := &percentage;
BEGIN
  raise_salary(v_deptno, v_percent);
END;
/


-- ============================================================
-- CALL THE PROCEDURE FROM THE COMMAND LINE
-- ============================================================
EXECUTE raise_salary(10, 10);
